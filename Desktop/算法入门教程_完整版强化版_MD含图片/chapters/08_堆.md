# 堆

## 什么是堆

堆是一种特殊的树形数据结构，满足堆性质。堆通常使用完全二叉树来实现，并且可以用数组高效存储。

### 堆的特性

1. **完全二叉树**：除了最后一层，其他层都被完全填满
2. **堆性质**：父节点与子节点之间满足特定关系
3. **数组存储**：可以用数组高效存储和访问

### 堆的类型

#### 最大堆（Max Heap）

- 每个节点的值都大于或等于其子节点的值
- 根节点是最大值

#### 最小堆（Min Heap）

- 每个节点的值都小于或等于其子节点的值
- 根节点是最小值

## 堆的数组表示

堆可以用数组来表示，对于索引为 i 的节点：
- 父节点索引：(i - 1) // 2
- 左子节点索引：2 * i + 1
- 右子节点索引：2 * i + 2

```
数组: [10, 9, 8, 7, 6, 5, 4]
堆结构:
        10
       /  \
      9    8
     / \  / \
    7  6 5  4
```

## 堆的基本操作

### 堆化（Heapify）

将一个无序数组调整为堆结构。

```python
def heapify(arr, n, i):
    """堆化操作（最大堆）"""
    largest = i
    left = 2 * i + 1
    right = 2 * i + 2
    
    # 检查左子节点是否大于当前最大值
    if left < n and arr[left] > arr[largest]:
        largest = left
    
    # 检查右子节点是否大于当前最大值
    if right < n and arr[right] > arr[largest]:
        largest = right
    
    # 如果最大值不是当前节点，交换并继续堆化
    if largest != i:
        arr[i], arr[largest] = arr[largest], arr[i]
        heapify(arr, n, largest)
```

### 构建堆

```python
def build_heap(arr):
    """构建最大堆"""
    n = len(arr)
    
    # 从最后一个非叶子节点开始堆化
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)
    
    return arr

# 使用示例
arr = [4, 10, 3, 5, 1]
build_heap(arr)
print(arr)  # [10, 5, 3, 4, 1]
```

### 插入元素

```python
def heap_insert(heap, item):
    """向堆中插入元素"""
    heap.append(item)
    i = len(heap) - 1
    
    # 向上调整
    while i > 0:
        parent = (i - 1) // 2
        if heap[parent] < heap[i]:  # 最大堆
            heap[parent], heap[i] = heap[i], heap[parent]
            i = parent
        else:
            break
    
    return heap

# 使用示例
heap = [10, 5, 3, 4, 1]
heap_insert(heap, 8)
print(heap)  # [10, 8, 3, 4, 1, 5]
```

### 提取最大值（最大堆）

```python
def heap_extract_max(heap):
    """提取最大值"""
    if not heap:
        raise IndexError("堆为空")
    
    if len(heap) == 1:
        return heap.pop()
    
    max_val = heap[0]
    heap[0] = heap.pop()
    
    # 向下调整
    i = 0
    n = len(heap)
    
    while True:
        left = 2 * i + 1
        right = 2 * i + 2
        largest = i
        
        if left < n and heap[left] > heap[largest]:
            largest = left
        
        if right < n and heap[right] > heap[largest]:
            largest = right
        
        if largest != i:
            heap[i], heap[largest] = heap[largest], heap[i]
            i = largest
        else:
            break
    
    return max_val

# 使用示例
heap = [10, 8, 3, 4, 1, 5]
print(heap_extract_max(heap))  # 10
print(heap)  # [8, 5, 3, 4, 1]
```

## 堆的实现

### 最大堆实现

```python
class MaxHeap:
    def __init__(self):
        self.heap = []
    
    def parent(self, i):
        """获取父节点索引"""
        return (i - 1) // 2
    
    def left_child(self, i):
        """获取左子节点索引"""
        return 2 * i + 1
    
    def right_child(self, i):
        """获取右子节点索引"""
        return 2 * i + 2
    
    def insert(self, val):
        """插入元素"""
        self.heap.append(val)
        self._sift_up(len(self.heap) - 1)
    
    def _sift_up(self, i):
        """向上调整"""
        while i > 0 and self.heap[self.parent(i)] < self.heap[i]:
            self.heap[self.parent(i)], self.heap[i] = self.heap[i], self.heap[self.parent(i)]
            i = self.parent(i)
    
    def extract_max(self):
        """提取最大值"""
        if not self.heap:
            raise IndexError("堆为空")
        
        if len(self.heap) == 1:
            return self.heap.pop()
        
        max_val = self.heap[0]
        self.heap[0] = self.heap.pop()
        self._sift_down(0)
        
        return max_val
    
    def _sift_down(self, i):
        """向下调整"""
        n = len(self.heap)
        
        while True:
            left = self.left_child(i)
            right = self.right_child(i)
            largest = i
            
            if left < n and self.heap[left] > self.heap[largest]:
                largest = left
            
            if right < n and self.heap[right] > self.heap[largest]:
                largest = right
            
            if largest != i:
                self.heap[i], self.heap[largest] = self.heap[largest], self.heap[i]
                i = largest
            else:
                break
    
    def peek(self):
        """查看最大值"""
        if not self.heap:
            raise IndexError("堆为空")
        return self.heap[0]
    
    def size(self):
        """堆的大小"""
        return len(self.heap)
    
    def is_empty(self):
        """判断堆是否为空"""
        return len(self.heap) == 0
```

## Python 中的堆

Python 的 `heapq` 模块提供了堆操作：

```python
import heapq

# 创建堆
heap = []
heapq.heappush(heap, 3)
heapq.heappush(heap, 1)
heapq.heappush(heap, 4)
heapq.heappush(heap, 2)
print(heap)  # [1, 2, 4, 3] - 最小堆

# 弹出最小值
print(heapq.heappop(heap))  # 1
print(heap)  # [2, 3, 4]

# 查看最小值
print(heap[0])  # 2

# 将列表转换为堆
arr = [5, 3, 8, 1, 2]
heapq.heapify(arr)
print(arr)  # [1, 2, 8, 5, 3]

# 弹出 n 个最小值
print(heapq.nsmallest(2, arr))  # [1, 2]

# 弹出 n 个最大值
print(heapq.nlargest(2, arr))  # [8, 5]
```

## 堆的应用

### 应用1：堆排序

```python
def heap_sort(arr):
    """堆排序"""
    n = len(arr)
    
    # 构建最大堆
    for i in range(n // 2 - 1, -1, -1):
        heapify(arr, n, i)
    
    # 逐个提取元素
    for i in range(n - 1, 0, -1):
        arr[0], arr[i] = arr[i], arr[0]
        heapify(arr, i, 0)
    
    return arr

# 使用示例
arr = [12, 11, 13, 5, 6, 7]
print(heap_sort(arr))  # [5, 6, 7, 11, 12, 13]
```

### 应用2：优先队列

```python
class PriorityQueue:
    def __init__(self):
        self.heap = MinHeap()
    
    def enqueue(self, item, priority):
        """入队（带优先级）"""
        self.heap.insert((priority, item))
    
    def dequeue(self):
        """出队（优先级最高的）"""
        if self.heap.is_empty():
            raise IndexError("队列为空")
        return self.heap.extract_min()[1]
    
    def peek(self):
        """查看队首元素"""
        if self.heap.is_empty():
            raise IndexError("队列为空")
        return self.heap.peek()[1]
    
    def is_empty(self):
        """判断队列是否为空"""
        return self.heap.is_empty()

# 使用示例
pq = PriorityQueue()
pq.enqueue("任务1", 3)
pq.enqueue("任务2", 1)
pq.enqueue("任务3", 2)

print(pq.dequeue())  # 任务2（优先级最高）
print(pq.dequeue())  # 任务3
print(pq.dequeue())  # 任务1
```

### 应用3：Top K 问题

```python
import heapq

def top_k_elements(arr, k):
    """找出数组中最大的 k 个元素"""
    if k <= 0:
        return []
    if k >= len(arr):
        return arr
    
    # 使用最小堆保持 k 个最大元素
    heap = arr[:k]
    heapq.heapify(heap)
    
    for num in arr[k:]:
        if num > heap[0]:
            heapq.heappop(heap)
            heapq.heappush(heap, num)
    
    return sorted(heap, reverse=True)

# 使用示例
arr = [3, 2, 1, 5, 6, 4]
k = 3
print(top_k_elements(arr, k))  # [6, 5, 4]
```

### 应用4：合并有序数组

```python
import heapq

def merge_sorted_arrays(arrays):
    """合并多个有序数组"""
    heap = []
    result = []
    
    # 将每个数组的第一个元素放入堆中
    for i, arr in enumerate(arrays):
        if arr:
            heapq.heappush(heap, (arr[0], i, 0))
    
    while heap:
        val, array_idx, element_idx = heapq.heappop(heap)
        result.append(val)
        
        # 将同一数组的下一个元素放入堆中
        if element_idx + 1 < len(arrays[array_idx]):
            next_val = arrays[array_idx][element_idx + 1]
            heapq.heappush(heap, (next_val, array_idx, element_idx + 1))
    
    return result

# 使用示例
arrays = [
    [1, 4, 5],
    [1, 3, 4],
    [2, 6]
]
print(merge_sorted_arrays(arrays))  # [1, 1, 2, 3, 4, 4, 5, 6]
```

### 应用5：中位数维护

```python
import heapq

class MedianFinder:
    def __init__(self):
        self.max_heap = []  # 存储较小的一半（最大堆）
        self.min_heap = []  # 存储较大的一半（最小堆）
    
    def add_num(self, num):
        """添加数字"""
        # 先放入最大堆
        heapq.heappush(self.max_heap, -num)
        
        # 确保最大堆的最大值 <= 最小堆的最小值
        if self.min_heap and -self.max_heap[0] > self.min_heap[0]:
            val = -heapq.heappop(self.max_heap)
            heapq.heappush(self.min_heap, val)
        
        # 平衡两个堆的大小
        if len(self.max_heap) > len(self.min_heap) + 1:
            val = -heapq.heappop(self.max_heap)
            heapq.heappush(self.min_heap, val)
        elif len(self.min_heap) > len(self.max_heap):
            val = heapq.heappop(self.min_heap)
            heapq.heappush(self.max_heap, -val)
    
    def find_median(self):
        """查找中位数"""
        if not self.max_heap and not self.min_heap:
            raise ValueError("没有数据")
        
        if len(self.max_heap) == len(self.min_heap):
            return (-self.max_heap[0] + self.min_heap[0]) / 2
        else:
            return -self.max_heap[0]

# 使用示例
mf = MedianFinder()
mf.add_num(1)
mf.add_num(2)
print(mf.find_median())  # 1.5
mf.add_num(3)
print(mf.find_median())  # 2
```

## 堆的复杂度分析

| 操作 | 时间复杂度 | 空间复杂度 |
|------|------------|------------|
| 构建堆 | O(n) | O(1) |
| 插入 | O(log n) | O(1) |
| 提取极值 | O(log n) | O(1) |
| 查看极值 | O(1) | O(1) |
| 堆排序 | O(n log n) | O(1) |

## 堆 vs 其他数据结构

### 与数组对比

| 特性 | 堆 | 数组 |
|------|------|------|
| 查找极值 | O(1) | O(n) |
| 插入 | O(log n) | O(1) |
| 删除极值 | O(log n) | O(n) |
| 有序性 | 部分有序 | 完全有序/无序 |

### 与平衡树对比

| 特性 | 堆 | 平衡树 |
|------|------|--------|
| 查找极值 | O(1) | O(log n) |
| 插入 | O(log n) | O(log n) |
| 删除极值 | O(log n) | O(log n) |
| 查找任意值 | O(n) | O(log n) |
| 范围查询 | 不支持 | O(log n) |

## 堆的优缺点

### 优点

1. **快速获取极值**：O(1) 时间获取最大值或最小值
2. **高效插入删除**：O(log n) 时间插入和删除
3. **空间效率高**：使用数组存储，空间利用率高
4. **实现简单**：相对容易实现和理解

### 缺点

1. **查找效率低**：查找特定元素需要 O(n) 时间
2. **不支持范围查询**：无法高效进行范围查询
3. **顺序访问差**：无法按顺序遍历所有元素

## 总结

堆是一种重要的树形数据结构：

- **核心特性**：完全二叉树，满足堆性质
- **高效操作**：O(1) 获取极值，O(log n) 插入删除
- **数组存储**：可以用数组高效存储和访问
- **广泛应用**：堆排序、优先队列、Top K 问题等

堆在实际开发中应用广泛，特别是在需要频繁获取极值的场景中表现优异。
